ภูมิหลัง​

บทช่วยสอนนี้จำลองวิวัฒนาการตามเวลาแบบ Trotter ของแบบจำลอง Ising แบบ tilted-field ด้วยฟังก์ชัน LTM AI-QEM

แบบจำลอง Ising [1] เดิมคิดขึ้นเพื่อทำความเข้าใจเฟอร์โรแมกเนติซึมและการเปลี่ยนเฟสควอนตัมในวัสดุแม่เหล็ก และกลายเป็นรากฐานของฟิสิกส์หลายวัตถุสมัยใหม่ ในระดับพื้นฐานที่สุด แบบจำลองนี้อธิบายแลตทิซของสปินที่ทำอันตรกิริยากัน โดยสปินแต่ละตัวอยู่ในสถานะได้หนึ่งในสองสถานะและทำอันตรกิริยากับเพื่อนบ้านใกล้สุด พลวัตแบบเรียลไทม์ของระบบ Ising สองมิติยิ่งจำลองแบบคลาสสิกได้ยากขึ้นเมื่อขนาดระบบและเวลาวิวัฒนาการเพิ่มขึ้น จึงเป็นสนามทดสอบสำคัญสำหรับการจำลองควอนตัม นอกจากนี้ เนื่องจาก Hamiltonian ประกอบด้วยสนามหนึ่งคิวบิตแบบโลคัลและอันตรกิริยาสองคิวบิตระหว่างเพื่อนบ้านใกล้สุด วิวัฒนาการตามเวลาของมันจึงแทนด้วยลำดับของ Gate หนึ่งและสองคิวบิตได้อย่างเป็นธรรมชาติ คุณสมบัติเหล่านี้ทำให้แบบจำลองนี้เป็นเกณฑ์วัดที่มีประโยชน์สำหรับศึกษาพลวัตควอนตัมบนฮาร์ดแวร์ปัจจุบัน และสำหรับประเมินประสิทธิผลของวิธีลดข้อผิดพลาดควอนตัม

บทช่วยสอนนี้ครอบคลุมขั้นตอนต่อไปนี้:

  • อธิบายแบบจำลอง Ising แบบ tilted-field และวิวัฒนาการตามเวลาแบบ Trotter ทางคณิตศาสตร์

  • กำหนด Backend และคิวบิตที่จะใช้จำลอง Hamiltonian แล้วแสดงภาพ

  • กำหนด Hamiltonian ของ Ising แบบ tilted-field ด้วยคลาส SparsePauliOp ของ Qiskit

  • สร้าง Circuit ควอนตัมสำหรับวิวัฒนาการของ Hamiltonian ด้วยฟังก์ชัน evolved_operator_ansatz ของ Qiskit

  • กำหนด observable ที่สอดคล้องกับ magnetization เฉลี่ย โดยใช้คลาส SparsePauliOp อีกครั้ง

  • คำนวณวิวัฒนาการของ magnetization เฉลี่ยสำหรับจำนวน Trotter step ที่เพิ่มขึ้น ด้วยฟังก์ชัน LTM AI-QEM

  • ใช้วิธี zero-noise extrapolation (ZNE) กับผลลัพธ์ดิบจาก QPU เพื่อเปรียบเทียบกับการลดข้อผิดพลาดที่ฟังก์ชัน LTM AI-QEM ทำ

  • แสดงรายการค่า magnetization เฉลี่ยที่คำนวณด้วยอัลกอริทึม sparse Pauli dynamics (SPD) ซึ่งใช้เป็นค่าในอุดมคติ

  • แสดงภาพผลลัพธ์ทั้งหมดเพื่อเปรียบเทียบ magnetization เฉลี่ยที่ลดข้อผิดพลาดด้วย LTM AI-QEM กับผลที่ไม่ได้ลดข้อผิดพลาด ผลที่ลดข้อผิดพลาดด้วย ZNE และผลจากอัลกอริทึม SPD แบบคลาสสิก

def generate_trotter_step_circuit(ham, step):
    circuit = evolved_operator_ansatz(ham, reps=step, insert_barriers=False)  # type: ignore
    circuit.assign_parameters([1] * circuit.num_parameters, inplace=True)

    barrier_positions = [n * len(ham) for n in range(1, step + 1)]
    barrier_op = Barrier(circuit.num_qubits)
    barrier_instruction = CircuitInstruction(
        operation=barrier_op, qubits=circuit.qubits
    )
    for barrier_pos in sorted(barrier_positions, reverse=True):
        circuit.data.insert(barrier_pos, barrier_instruction)

    return circuit

import matplotlib.pyplot as plt
import numpy as np

def compute_improvement_metrics(
    spd_results, mitigated_evs, zne_evs, noisy_evs
):
    """Quantify how much each mitigation method improves on the unmitigated result.

    Error for each method is |SPD - method| at every Trotter step (SPD is ideal).

    Returns a dict keyed by method name with:
      - mae:               mean absolute error vs SPD
      - rmse:              root mean squared error vs SPD
      - mean_improvement:  mean(err_unmitigated - err_method)   (absolute units, >0 is better)
      - pct_improvement:   100 * (MAE_unmitigated - MAE_method) / MAE_unmitigated
      - error_reduction_factor: MAE_unmitigated / MAE_method    (>1 is better)
      - frac_steps_improved:   fraction of steps where the method beats unmitigated
    """
    spd = np.asarray(spd_results, dtype=float)
    errs = {
        "Unmitigated": np.abs(spd - np.asarray(noisy_evs, dtype=float)),
        "ZNE": np.abs(spd - np.asarray(zne_evs, dtype=float)),
        "LTM AI-QEM": np.abs(spd - np.asarray(mitigated_evs, dtype=float)),
    }
    base = errs["Unmitigated"]
    base_mae = base.mean()
    eps = 1e-12

    metrics = {}
    for name, e in errs.items():
        mae = e.mean()
        metrics[name] = {
            "mae": mae,
            "rmse": float(np.sqrt(np.mean(e**2))),
            "mean_improvement": float((base - e).mean()),
            "pct_improvement": float(
                100 * (base_mae - mae) / max(base_mae, eps)
            ),
            "error_reduction_factor": float(base_mae / max(mae, eps)),
        }
    return metrics

def plot_trotter_results(
    trotter_steps,
    spd_results,
    mitigated_evs,
    mitigated_errs,
    zne_evs,
    zne_errs,
    noisy_evs,
    noisy_errs,
    title="",
    print_summary=True,
):
    """Plot magnetization versus Trotter steps, and the absolute difference from SPD for each method.

    Top: average magnetization for each method.
    Bottom: grouped bars of |SPD - method|, with mean improvement annotated.

    Returns a dict of improvement metrics (see compute_improvement_metrics).
    """
    spd = np.asarray(spd_results)
    steps = np.asarray(trotter_steps)

    metrics = compute_improvement_metrics(
        spd_results, mitigated_evs, zne_evs, noisy_evs
    )

        nrows=2, ncols=1, figsize=(12, 10), sharex=True, height_ratios=[2, 1]
    )

            color=color,

            markeredgewidth=0.5,

            capsize=5,
        )

    ax1.set_title(title)
    ax1.grid(alpha=0.3)
    ax1.legend()

    # Bottom: grouped bars with MAE and % improvement in the legend labels
    bars = [
